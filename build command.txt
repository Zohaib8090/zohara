# 1. Clean build artifacts & clear package cache
sudo docker kill zohara-build 2>/dev/null || true
sudo rm -rf ~/Documents/my/zohara/out/ ~/Documents/my/zohara/work/ ~/Documents/my/zohara/pkg-cache/
mkdir -p ~/Documents/my/zohara/pkg-cache

# 2. Rebuild the builder Docker container (from scratch)
sudo docker build --no-cache -t zohara-builder ~/Documents/my/zohara

# 3. Generate the final Zohara OS ISO
sudo docker run --rm --name zohara-build \
  --privileged \
  -v ~/Documents/my/zohara:/build \
  -v ~/Documents/my/zohara/pkg-cache:/var/cache/pacman/pkg \
  zohara-builder
