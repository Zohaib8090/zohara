# 1. Clean previous build artifacts
sudo rm -rf ~/Documents/my/zohara/out/ ~/Documents/my/zohara/work/
sudo docker rm -f zohara-build 2>/dev/null

# 2. Rebuild docker builder image (registers the entrypoint runtime copy step)
sudo docker build -t zohara-builder .

# 3. Build the ISO
sudo docker run --rm --name zohara-build \
  --privileged \
  -v ~/Documents/my/zohara:/build \
  -v ~/Documents/my/zohara/pkg-cache:/var/cache/pacman/pkg \
  zohara-builder
